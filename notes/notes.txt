

TRUNCATE TABLE tb_settle;


INSERT INTO tb_settle (stl_dt, stl_status)
	SELECT 
		a.cl_dt stl_dt,
		1 stl_status
	FROM
		tb_shift a
	WHERE
	a.cl_id = (
		SELECT b.cl_id FROM tb_shift b WHERE
		DATE(a.cl_dt) = DATE(b.cl_dt)
		ORDER BY b.cl_dt DESC LIMIT 1)
;

DELIMITER $$
DROP FUNCTION IF EXISTS `SeqNo`$$
CREATE FUNCTION `SeqNo`(_name TEXT) RETURNS INT(11)
BEGIN
   DECLARE seqno INT;
   UPDATE tx_seq SET seq_no = LAST_INSERT_ID(seq_no + 1) WHERE seq_name=_name;
   SELECT LAST_INSERT_ID() INTO seqno;
   RETURN seqno;
END$$
DELIMITER ;
